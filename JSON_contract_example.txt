##### MQTT Message Contract #####

1. MQTT message contract:

    a. provisioning_request

        topic:
            smartcampus/v1/device/provision/request

        usage:
            Register a newly booted ESP32 device without completed configuration.

        content:
        {
            "message_id": "uuid-string",
            "source_timestamp": "ISO8601-datetime-string",
            "payload":
            {
                "mac_address": "string (format: XX:XX:XX:XX:XX:XX)",
                "firmware_version": "string (e.g. 1.0.0)",
                "hardware_revision": "string (e.g. ESP32-WROOM-32E)"
            }
        }


    b. provisioning_response

        topic:
            smartcampus/v1/device/provision/response/{mac_address}

        usage:
            Return provisioning information for a registered device.

        content:
        {
            "message_id": "uuid-string",
            "source_timestamp": "ISO8601-datetime-string",
            "payload":
            {
                "device_id": "uuid-string",
                "device_name": "string",
                "room_id": "uuid-string",
                "room_type": "string (values: classroom | corridor)",
                "firmware_version": "string",
                "heartbeat_interval": 30
            }
        }

        notes:
        - hard priority: device must be registered before any other action.


    c. device_heartbeat

        topic_subscribe:
            smartcampus/v1/device/+/heartbeat

        topic_publish:
            smartcampus/v1/device/{device_id}/heartbeat

        usage:
            Check device online status.

        content:
        {
            "message_id": "uuid-string",
            "source_timestamp": "ISO8601-datetime-string",
            "payload":
            {
                "device_id": "uuid-string",
                "alive": true,
                "firmware_version": "string",
                "uptime": 0
            }
        }

        notes:
        - last_seen is generated by the server.
        - alive is boolean.


    d. environment_telemetry

        topic_subscribe:
            smartcampus/v1/telemetry/room/+/environment

        topic_publish:
            smartcampus/v1/telemetry/room/{room_id}/environment
        
        usage:
            Sending timeseries environment telemetry stats.

        content:
        {
            "message_id": "uuid-string",
            "source_timestamp": "ISO8601-datetime-string",
            "payload":
            {
                "room_id": "uuid-string",
                "temperature": 0.0,
                "humidity": 0.0,
                "smoke_detected": false,
                "smoke_value": 0.0,
                "smoke_threshold": 0.0,
                "smoke_state": "string (values: normal | suspected | emergency)",
                "co2": 0,
                "air_quality": 0,
                "room_mode": "string (values: saving | self_study | lecture | exam | lock | suspected | emergency)"
            }
        }


    e. occupancy_telemetry

        topic_subscribe:
            smartcampus/v1/telemetry/room/+/occupancy

        topic_publish:
            smartcampus/v1/telemetry/room/{room_id}/occupancy

        usage:
            Sending timeseries occupancy telemetry stats from IR motion sensors.

        content:
        {
            "message_id": "uuid-string",
            "source_timestamp": "ISO8601-datetime-string",
            "payload":
            {
                "room_id": "uuid-string",
                "occupancy_type": "string (values: in | out)",
                "occupancy_count": 0
            }
        }


    f. room_rfid_event

        topic_subscribe:
            smartcampus/v1/event/room/+/rfid

        topic_publish:
            smartcampus/v1/event/room/{room_id}/rfid

        usage:
            Report RFID card tap event at classroom/room entrance for attendance tracking.

        content:
        {
            "message_id": "uuid-string",
            "source_timestamp": "ISO8601-datetime-string",
            "payload":
            {
                "room_id": "uuid-string",
                "card_uid": "string",
                "event_type": "string (values: check_in | check_out | reject)"
            }
        }


    g. room_state

        topic_publish:
            smartcampus/v1/room/{room_id}/state

        usage:
            Publish room mode and status changes to devices in the room (sent by gateway/server).

        content:
        {
            "message_id": "uuid-string",
            "source_timestamp": "ISO8601-datetime-string",
            "payload":
            {
                "room_id": "uuid-string",
                "room_mode": "string (values: saving | self_study | lecture | exam | lock | suspected | emergency)",
                "room_status": "string (values: online | offline)"
            }
        }


    h. device_status

        topic_publish:
            smartcampus/v1/device/{mac_address}/status

        usage:
            Publish device connection status change (sent by gateway/server).

        content:
        {
            "message_id": "uuid-string",
            "source_timestamp": "ISO8601-datetime-string",
            "payload":
            {
                "device_id": "uuid-string",
                "mac_address": "string (format: XX:XX:XX:XX:XX:XX)",
                "device_status": "string (values: online | offline)",
                "reason": "string"
            }
        }


    i. room_command

        topic_publish:
            smartcampus/v1/command/room/{room_id}

        usage:
            Control peripherals inside a classroom or room.

        content:
        {
            "message_id": "uuid-string",
            "source_timestamp": "ISO8601-datetime-string",
            "payload":
            {
                "room_id": "uuid-string",
                "command_type": "string (values: mode | door | fan | light | buzzer)",
                "command_value": "string/number/boolean"
            }
        }


    j. device_command

        topic_publish:
            smartcampus/v1/command/device/{mac_address}

        usage:
            Send maintenance or configuration commands to a specific device.

        content:
        {
            "message_id": "uuid-string",
            "source_timestamp": "ISO8601-datetime-string",
            "payload":
            {
                "mac_address": "string (format: XX:XX:XX:XX:XX:XX)",
                "command_type": "string (values: ota | restart)",
                "command_value": "string"
            }
        }


    k. command_ack

        topic_subscribe:
            smartcampus/v1/ack/device/+/command/+

        topic_publish:
            smartcampus/v1/ack/device/{mac_address}/command/{command_id}

        usage:
            Acknowledge receipt and execution status of a room or device command.

        content:
        {
            "message_id": "uuid-string",
            "source_timestamp": "ISO8601-datetime-string",
            "payload":
            {
                "mac_address": "string (format: XX:XX:XX:XX:XX:XX)",
                "command_id": "uuid-string",
                "success": true,
                "error_message": "string"
            }
        }


    l. scenario

        topic_subscribe:
            smartcampus/v1/scenario/#

        usage:
            Trigger or notify state updates of automation scenarios across the campus.

        content:
        {
            "message_id": "uuid-string",
            "source_timestamp": "ISO8601-datetime-string",
            "payload":
            {
                "scenario_id": "uuid-string",
                "action": "string (e.g. execute | cancel)",
                "triggered_by": "string"
            }
        }
